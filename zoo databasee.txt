/* =====================================================
   CREATE DATABASE
   This creates a new database called MovieStudioDB
===================================================== */
CREATE DATABASE MovieStudioDB;
GO

/* Switch to the new database */
USE MovieStudioDB;
GO

/* =====================================================
   CREATE DIRECTORS TABLE (Parent Table)
   DirectorID is the PRIMARY KEY
===================================================== */
CREATE TABLE Directors (
    DirectorID INT PRIMARY KEY,        -- Unique identifier for each director
    DirectorName VARCHAR(100),         -- Director's full name
    MovieTitle VARCHAR(100)            -- Movie they directed
);
GO

/* =====================================================
   CREATE ACTORS TABLE (Child Table)
   ActorID is PRIMARY KEY
   DirectorID is FOREIGN KEY referencing Directors
===================================================== */
CREATE TABLE Actors (
    ActorID INT PRIMARY KEY,           -- Unique identifier for each actor
    ActorName VARCHAR(100),            -- Actor's full name
    MovieTitle VARCHAR(100),           -- Movie the actor appeared in
    DirectorID INT,                    -- Foreign key linking to Directors table

    CONSTRAINT FK_Director
    FOREIGN KEY (DirectorID)
    REFERENCES Directors(DirectorID)
);
GO

/* =====================================================
   QUERY DATA FROM BOTH TABLES
   Join Actors and Directors using DirectorID
   Shows shared MovieTitle attribute
===================================================== */
SELECT 
    Actors.ActorName,
    Directors.DirectorName,
    Actors.MovieTitle
FROM Actors
INNER JOIN Directors
ON Actors.DirectorID = Directors.DirectorID;
GO
